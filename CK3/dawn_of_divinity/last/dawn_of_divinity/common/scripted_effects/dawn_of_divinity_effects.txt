# Выдать один драфт силы, если ни один драфт ещё не открыт
dod_add_draft = {
	if = {
		limit = { NOT = { has_character_flag = dod_draft_open } }
		add_character_flag = dod_draft_open
		trigger_event = { id = dod_power_events.0001 days = 1 }
	}
}
