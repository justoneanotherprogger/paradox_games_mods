# Dawn of Divinity — story cycle носителя (структура автора, вызовы исправлены)
story_cycle_main_character_roguelike_dod = {

	on_setup = {
		set_variable = {
			name = prestige_threshold_for_power
			value = 100
		}
		set_variable = {
			name = piety_threshold_for_power
			value = 100
		}
		story_owner = {
			add_character_flag = roguelike_dod_story_created
			set_initial_values_for_roguelike_dod_modifiers = yes
		}
	}

	on_end = {
		story_owner = {
			remove_dod_roguelike_story_modifiers = yes
		}
	}

	on_owner_death = {
		scope:story = { end_story = yes }
	}

	effect_group = {
		days = 10
		chance = 100

		trigger = {
			exists = var:prestige_threshold_for_power
			story_owner.prestige >= var:prestige_threshold_for_power
		}

		triggered_effect = {
			trigger = { always = yes }
			effect = {
				change_variable = {
					name = prestige_threshold_for_power
					multiply = 1.5
				}
				story_owner = {
					dod_add_draft = yes
				}
			}
		}
	}

	effect_group = {
		days = 10
		chance = 100

		trigger = {
			exists = var:piety_threshold_for_power
			story_owner.piety >= var:piety_threshold_for_power
		}

		triggered_effect = {
			trigger = { always = yes }
			effect = {
				change_variable = {
					name = piety_threshold_for_power
					multiply = 1.5
				}
				story_owner = {
					dod_add_draft = yes
				}
			}
		}
	}

	effect_group = {
		months = { 12 36 }
		chance = 100

		triggered_effect = {
			trigger = { always = yes }
			effect = {
				story_owner = {
					dod_add_draft = yes
				}
			}
		}
	}
}
